import React from "react";

import { isAuthenticated, getUserRole } from "../utils/auth";

interface AdminRouteProps {
  children?: React.ReactNode;
}

const AdminRoute: React.FC<AdminRouteProps> = ({ children }) => {
  if (!isAuthenticated()) {
    return <Navigate to="/" replace />;
  }

  const role = getUserRole();
  if (role !== "ADMIN") {
    return <Navigate to="/dashboard/employees" replace />;
  }

  return children ? <>{children}</> : <Outlet />;
};

import { useState, useEffect, type FormEvent } from "react";
import axios from "axios";
import API from "../api/api";
import { getUserRole } from "../utils/auth";
import "../styles/dashboard.css";

interface UserItem {
  id: number;
  username: string;
  role: "ADMIN" | "USER";
}

interface ApiErrorData {
  message?: string;
  error?: string;
  errors?: Record<string, string>;
}

function ManageUsers() {
  const [users, setUsers] = useState<UserItem[]>([]);
  const [loading, setLoading] = useState(false);
  const [listError, setListError] = useState<string | null>(null);

  // New user form state
  const [username, setUsername] = useState("");
  const [password, setPassword] = useState("");
  const [role, setRole] = useState<"ADMIN" | "USER">("USER");
  const [submitting, setSubmitting] = useState(false);
  const [formError, setFormError] = useState<string | null>(null);
  const [formSuccess, setFormSuccess] = useState<string | null>(null);

  const currentUserRole = getUserRole();
  const isAdmin = currentUserRole === "ADMIN";

  const fetchUsers = async () => {
    try {
      setLoading(true);
      setListError(null);
      const res = await API.get<UserItem[]>("/users");
      setUsers(Array.isArray(res.data) ? res.data : []);
    } catch (err: unknown) {
      let msg = "Failed to load users.";
      if (axios.isAxiosError<ApiErrorData>(err)) {
        msg = err.response?.data?.message || err.response?.data?.error || msg;
      }
      setListError(msg);
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    if (isAdmin) {
      fetchUsers();
    }
  }, [isAdmin]);

  if (!isAdmin) {
    return (
      <div className="dashboardContent">
        <div className="card">
          <div className="alert alertError">
            Access Denied: Only administrators have permission to manage users.
          </div>
        </div>
      </div>
    );
  }

  const handleCreateUser = async (e: FormEvent) => {
    e.preventDefault();
    setFormError(null);
    setFormSuccess(null);

    const trimmedUsername = username.trim();
    if (!trimmedUsername) {
      setFormError("Username cannot be blank.");
      return;
    }

    if (trimmedUsername.length < 3 || trimmedUsername.length > 50) {
      setFormError("Username must be between 3 and 50 characters.");
      return;
    }

    if (!password) {
      setFormError("Password cannot be blank.");
      return;
    }

    if (password.length < 6) {
      setFormError("Password must be at least 6 characters.");
      return;
    }

    try {
      setSubmitting(true);
      await API.post("/users", {
        username: trimmedUsername,
        password: password,
        role: role,
      });

      setFormSuccess(`User "${trimmedUsername}" created successfully with role ${role}!`);
      setUsername("");
      setPassword("");
      setRole("USER");
      await fetchUsers();
    } catch (err: unknown) {
      let msg = "Failed to create user.";
      if (axios.isAxiosError<ApiErrorData>(err)) {
        if (err.response?.data?.errors) {
          const errors = err.response.data.errors;
          const firstKey = Object.keys(errors)[0];
          msg = errors[firstKey];
        } else if (err.response?.data?.message) {
          msg = err.response.data.message;
        } else if (err.response?.data?.error) {
          msg = err.response.data.error;
        }
      } else if (err instanceof Error) {
        msg = err.message;
      }
      setFormError(msg);
    } finally {
      setSubmitting(false);
    }
  };

  return (
    <div className="dashboardContent">
      {/* CREATE NEW USER CARD */}
      <div className="card" style={{ marginBottom: "30px" }}>
        <div className="cardHeader">
          <h2 className="cardTitle">Create New User</h2>
          <p className="cardSubtitle">Add a new portal user and assign their access role</p>
        </div>

        {formSuccess && (
          <div className="alert alertSuccess">{formSuccess}</div>
        )}

        {formError && (
          <div className="alert alertError">{formError}</div>
        )}

        <form onSubmit={handleCreateUser}>
          <div style={{ marginBottom: "15px" }}>
            <label className="formLabel">Username</label>
            <input
              className="inputField"
              placeholder="Username (3-50 characters)"
              value={username}
              onChange={(e) => setUsername(e.target.value)}
              disabled={submitting}
              required
            />
          </div>

          <div style={{ marginBottom: "15px" }}>
            <label className="formLabel">Password</label>
            <input
              type="password"
              className="inputField"
              placeholder="Password (min 6 characters)"
              value={password}
              onChange={(e) => setPassword(e.target.value)}
              disabled={submitting}
              required
            />
          </div>

          <div style={{ marginBottom: "20px" }}>
            <label className="formLabel">Role</label>
            <select
              className="selectField"
              value={role}
              onChange={(e) => setRole(e.target.value as "ADMIN" | "USER")}
              disabled={submitting}
            >
              <option value="USER">USER (Standard Employee Access)</option>
              <option value="ADMIN">ADMIN (Full Administrative Access)</option>
            </select>
          </div>

          <button
            type="submit"
            className="primaryButton"
            disabled={submitting}
            style={{ width: "100%" }}
          >
            {submitting ? "Creating User..." : "Create User"}
          </button>
        </form>
      </div>

      {/* ALL USERS LIST CARD */}
      <div className="card">
        <div className="cardHeader">
          <h2 className="cardTitle">System Users</h2>
          <p className="cardSubtitle">All registered user accounts</p>
        </div>

        {listError && <div className="alert alertError">{listError}</div>}

        <div className="tableResponsive">
          <table className="employeeTable">
            <thead>
              <tr>
                <th>ID</th>
                <th>Username</th>
                <th>Role</th>
              </tr>
            </thead>

            <tbody>
              {loading ? (
                <tr>
                  <td colSpan={3} style={{ textAlign: "center", padding: "30px" }}>
                    Loading users...
                  </td>
                </tr>
              ) : users.length === 0 ? (
                <tr>
                  <td colSpan={3} style={{ textAlign: "center", padding: "30px", opacity: 0.7 }}>
                    No users found.
                  </td>
                </tr>
              ) : (
                users.map((u) => (
                  <tr key={u.id}>
                    <td>#{u.id}</td>
                    <td style={{ fontWeight: 600 }}>{u.username}</td>
                    <td>
                      <span
                        className={`badge ${
                          u.role === "ADMIN" ? "badgeAdmin" : "badgeUser"
                        }`}
                      >
                        {u.role}
                      </span>
                    </td>
                  </tr>
                ))
              )}
            </tbody>
          </table>
        </div>
      </div>
    </div>
  );
}

export default ManageUsers;

import { useState, useEffect, type FormEvent } from "react";
import axios from "axios";
import API from "../api/api";
import "../styles/dashboard.css";

interface UserProfile {
  id: number;
  username: string;
  role: string;
}

interface ApiErrorData {
  message?: string;
  error?: string;
  errors?: Record<string, string>;
}

function Profile() {
  const [profile, setProfile] = useState<UserProfile | null>(null);
  const [profileLoading, setProfileLoading] = useState(true);
  const [profileError, setProfileError] = useState<string | null>(null);

  // Change password form state
  const [currentPassword, setCurrentPassword] = useState("");
  const [newPassword, setNewPassword] = useState("");
  const [confirmPassword, setConfirmPassword] = useState("");
  const [passwordLoading, setPasswordLoading] = useState(false);
  const [passwordError, setPasswordError] = useState<string | null>(null);
  const [passwordSuccess, setPasswordSuccess] = useState<string | null>(null);

  const fetchProfile = async () => {
    try {
      setProfileLoading(true);
      setProfileError(null);
      const res = await API.get<UserProfile>("/users/me");
      setProfile(res.data);
    } catch (err: unknown) {
      let msg = "Failed to load user profile.";
      if (axios.isAxiosError<ApiErrorData>(err)) {
        msg = err.response?.data?.message || err.response?.data?.error || msg;
      }
      setProfileError(msg);
    } finally {
      setProfileLoading(false);
    }
  };

  useEffect(() => {
    fetchProfile();
  }, []);

  const handleChangePassword = async (e: FormEvent) => {
    e.preventDefault();
    setPasswordError(null);
    setPasswordSuccess(null);

    if (!currentPassword) {
      setPasswordError("Current password cannot be blank.");
      return;
    }

    if (!newPassword) {
      setPasswordError("New password cannot be blank.");
      return;
    }

    if (newPassword.length < 6) {
      setPasswordError("New password must be at least 6 characters.");
      return;
    }

    if (newPassword !== confirmPassword) {
      setPasswordError("New password and confirm password do not match.");
      return;
    }

    try {
      setPasswordLoading(true);
      const res = await API.put("/users/change-password", {
        currentPassword,
        newPassword,
      });

      const msg = res.data?.message || "Password changed successfully.";
      setPasswordSuccess(msg);
      setCurrentPassword("");
      setNewPassword("");
      setConfirmPassword("");
    } catch (err: unknown) {
      let msg = "Failed to change password.";
      if (axios.isAxiosError<ApiErrorData>(err)) {
        if (err.response?.data?.errors) {

import { useState, useEffect, type FormEvent } from "react";
import { useNavigate, Link } from "react-router-dom";
import axios from "axios";
import API from "../api/api";
import { isAuthenticated } from "../utils/auth";
import "../styles/login.css";

interface ApiErrorData {
  message?: string;
  error?: string;
  errors?: Record<string, string>;
}

function SignUp() {
  const navigate = useNavigate();

  const [username, setUsername] = useState("");
  const [password, setPassword] = useState("");
  const [confirmPassword, setConfirmPassword] = useState("");
  const [loading, setLoading] = useState(false);
  const [errorMessage, setErrorMessage] = useState<string | null>(null);

  useEffect(() => {
    if (isAuthenticated()) {
      navigate("/dashboard/employees", { replace: true });
    }
  }, [navigate]);

  const handleSignUp = async (e: FormEvent) => {
    e.preventDefault();
    setErrorMessage(null);

    const trimmedUsername = username.trim();
    if (!trimmedUsername) {
      setErrorMessage("Username cannot be blank.");
      return;
    }

    if (trimmedUsername.length < 3 || trimmedUsername.length > 50) {
      setErrorMessage("Username must be between 3 and 50 characters.");
      return;
    }

    if (!password) {
      setErrorMessage("Password cannot be blank.");
      return;
    }

    if (password.length < 6) {
      setErrorMessage("Password must be at least 6 characters.");
      return;
    }

    if (password !== confirmPassword) {
      setErrorMessage("Passwords do not match.");
      return;
    }

    try {
      setLoading(true);
      await API.post("/auth/register", {
        username: trimmedUsername,
        password: password,
      });

      // Redirect to login on success with success notification
      navigate("/", {
        state: { message: "Account created successfully! Please sign in." },
      });
    } catch (err: unknown) {
      let message = "Registration failed. Please try again.";
      if (axios.isAxiosError<ApiErrorData>(err)) {
        if (err.response?.data?.errors) {
          const errors = err.response.data.errors;
          const firstErrorKey = Object.keys(errors)[0];
          message = errors[firstErrorKey];
        } else if (err.response?.data?.message) {
          message = err.response.data.message;
        } else if (err.response?.data?.error) {
          message = err.response.data.error;
        } else if (err.message) {
          message = err.message;
        }
      } else if (err instanceof Error) {
        message = err.message;
      }
      setErrorMessage(message);
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="loginContainer">
      <div className="card loginCard">
        <h1 className="loginTitle">Create Account</h1>
        <p className="loginSubtitle">Sign up for Employee Portal access</p>

        {errorMessage && <div className="alert alertError">{errorMessage}</div>}

        <form onSubmit={handleSignUp}>
          <input
            className="inputField"
            placeholder="Username"
            value={username}
            onChange={(e) => setUsername(e.target.value)}
            disabled={loading}
            autoComplete="username"
            required
          />

          <input
            type="password"
            className="inputField"
            placeholder="Password (min 6 characters)"
            value={password}
            onChange={(e) => setPassword(e.target.value)}
            disabled={loading}
            autoComplete="new-password"
            required
          />

          <input
            type="password"
            className="inputField"
            placeholder="Confirm Password"
            value={confirmPassword}
            onChange={(e) => setConfirmPassword(e.target.value)}
            disabled={loading}
            autoComplete="new-password"
            required
          />

          <button
            type="submit"
            className="primaryButton"
            style={{ width: "100%" }}
            disabled={loading}
          >
            {loading ? "Creating Account..." : "Sign Up"}
          </button>
        </form>

        <div className="authLinkContainer">
          <span>Already have an account? </span>
          <Link to="/" className="authLink">
            Sign In
          </Link>
        </div>
      </div>
    </div>
  );
}

export default SignUp;

import { jwtDecode } from "jwt-decode";

export interface DecodedToken {
  sub?: string;
  role?: string;
  exp?: number;
  [key: string]: unknown;
}

export const getDecodedToken = (): DecodedToken | null => {
  const token = localStorage.getItem("token");
  if (!token) return null;

  try {
    const decoded = jwtDecode<DecodedToken>(token);
    if (decoded.exp && decoded.exp * 1000 < Date.now()) {
      localStorage.clear();
      return null;
    }
    return decoded;
  } catch {
    localStorage.clear();
    return null;
  }
};

export const isAuthenticated = (): boolean => {
  return getDecodedToken() !== null;
};

export const getUserRole = (): string | null => {
  const decoded = getDecodedToken();
  return decoded?.role ?? null;
};

export const getUsername = (): string | null => {
  const decoded = getDecodedToken();
  return decoded?.sub ?? null;
};

export const logoutUser = () => {
  localStorage.clear();
  window.location.href = "/";
};

# Employee Portal - Frontend SPA

A modern, responsive, enterprise-grade Employee Management Single Page Application (SPA) built with **React 19**, **TypeScript**, and **Vite**. Features Role-Based Access Control (RBAC), secure JWT authentication, paginated & searchable data views, inline editing, user management, and multi-stage Docker deployment.

---

## Tech Stack

| Category | Technology |
|---|---|
| **Core Framework** | [React 19](https://react.dev/) & [TypeScript](https://www.typescriptlang.org/) |
| **Build Tool** | [Vite 7](https://vitejs.dev/) |
| **Routing & Protection** | [React Router v7](https://reactrouter.com/) |
| **HTTP Client** | [Axios](https://axios-http.com/) (with Request & Response Interceptors) |
| **Authentication** | JWT Decode (`jwt-decode`), Bearer Token, Auto-Expiry Detection |
| **Styling** | [Tailwind CSS](https://tailwindcss.com/) & Custom CSS (Glassmorphism Dark Theme) |
| **Production Server** | [Nginx Alpine](https://hub.docker.com/_/nginx) with SPA Fallback |
| **Containerization** | Docker Multi-Stage Build |

---

## Key Features

### 1. Role-Based Access Control (RBAC) & Navigation
- **Dynamic UI Adapting to Roles**: Non-admin users (`USER`) have read-only access to employee records and their own profile. Administrative users (`ADMIN`) receive exclusive access to:
  - Add new employees (`/dashboard/register`)
  - Edit existing employee inline records
  - Delete employees with confirmation dialogs
  - Access system user management (`/dashboard/users`)
- **Active Navigation Indicator**: Sidebar dynamically highlights the active route using React Router `NavLink`. Restricted routes are automatically hidden from unauthorized users.

### 2. Comprehensive Route Protection
- **`ProtectedRoute`**: Inspects `localStorage` for a JWT token, decodes payload using `jwt-decode`, and validates expiration (`exp`). Redirects unauthenticated or expired visitors to `/`.
- **`AdminRoute`**: Enforces strict privilege checks, redirecting non-admin users to `/dashboard/employees`.
- **Index Auto-Redirect**: Automatically redirects `/dashboard` to `/dashboard/employees` to eliminate blank dashboard states.

### 3. Paginated & Searchable Employee Directory
- **Spring Data Pagination Integration**: Connects to `GET /employees/page?page={page}&size=5` with reactive Previous/Next controls, total element count, and dynamic page indicator.
- **Multi-Field Instant Search**: Filters records on the fly by employee name, email, or department.
- **Inline Editing**: Allows administrators to update employee information in place with strict client-side validation.
- **Delete Confirmation Modal**: Replaces basic browser popups with an accessible, dedicated confirmation modal wired to `DELETE /employees/{id}`.

### 4. User Profile & Password Management
- **User Profile Details**: Fetches authenticated user information from `GET /users/me` (Username, User ID, Role badge).
- **Password Update Form**: Secure change password flow via `PUT /users/change-password` with current password verification, 6+ character constraints, confirmation matching, and clear feedback banners.

### 5. Admin User Management
- **User Directory**: Lists all registered portal accounts (`GET /users`) with role badges.
- **User Provisioning**: Form to create new user accounts (`POST /users`) with customizable role assignment (`ADMIN` or `USER`).

### 6. Resilient API Client & Error Handling
- **Environment-Driven Base URL**: Loads `baseURL` from `import.meta.env.VITE_API_URL` with a sensible fallback.
- **Authorization Interceptor**: Automatically attaches `Bearer <token>` to all outgoing requests.
- **Session Auto-Logout**: Response interceptor intercepts `401 Unauthorized` responses, flushes `localStorage`, and redirects to `/`.

The application includes a production-ready **multi-stage Docker build**:

import { useEffect, useState, useCallback } from "react";
import axios from "axios";
import API from "../api/api";
import { getUserRole } from "../utils/auth";
import "../styles/dashboard.css";

interface Employee {
  id: number;
  name: string;
  email: string;
  department: string;
}

interface SpringPage<T> {
  content: T[];
  totalPages: number;
  totalElements: number;
  number: number;
  size: number;
  first: boolean;
  last: boolean;
}

interface ApiErrorData {
  message?: string;
  error?: string;
  errors?: Record<string, string>;
}

function Employees() {
  const [employees, setEmployees] = useState<Employee[]>([]);
  const [loading, setLoading] = useState(false);
  const [errorMessage, setErrorMessage] = useState<string | null>(null);
  const [successMessage, setSuccessMessage] = useState<string | null>(null);

  // Pagination state (0-indexed for Spring Data)
  const [page, setPage] = useState(0);
  const [totalPages, setTotalPages] = useState(1);
  const [totalElements, setTotalElements] = useState(0);
  const pageSize = 5;

  // Search filter
  const [searchTerm, setSearchTerm] = useState("");

  // Edit state
  const [editId, setEditId] = useState<number | null>(null);
  const [editEmployee, setEditEmployee] = useState({
    name: "",
    email: "",
    department: "",
  });
  const [isSaving, setIsSaving] = useState(false);

  // Delete modal state
  const [deleteTarget, setDeleteTarget] = useState<Employee | null>(null);
  const [isDeleting, setIsDeleting] = useState(false);

  const role = getUserRole();
  const isAdmin = role === "ADMIN";

  const loadEmployees = useCallback(
    async (pageToLoad: number) => {
      try {
        setLoading(true);
        setErrorMessage(null);
        const res = await API.get<SpringPage<Employee>>(
          `/employees/page?page=${pageToLoad}&size=${pageSize}`
        );

        const pageData = res.data;
        if (pageData && Array.isArray(pageData.content)) {
          setEmployees(pageData.content);
          setTotalPages(pageData.totalPages || 1);
          setTotalElements(pageData.totalElements || 0);
        } else if (Array.isArray(pageData)) {
          setEmployees(pageData);
          setTotalPages(1);
          setTotalElements((pageData as Employee[]).length);
        } else {
          setEmployees([]);
          setTotalPages(1);
          setTotalElements(0);
        }
      } catch (err: unknown) {
        let msg = "Failed to load employees.";
        if (axios.isAxiosError<ApiErrorData>(err)) {
          msg = err.response?.data?.message || err.response?.data?.error || msg;
        }
        setErrorMessage(msg);
      } finally {
        setLoading(false);
      }
    },
    [pageSize]
  );

  useEffect(() => {
    loadEmployees(page);
  }, [loadEmployees, page]);

  const startEdit = (emp: Employee) => {
    setEditId(emp.id);
    setEditEmployee({
      name: emp.name,
      email: emp.email,
      department: emp.department,
    });
    setErrorMessage(null);
    setSuccessMessage(null);
  };

  const cancelEdit = () => {
    setEditId(null);
    setEditEmployee({ name: "", email: "", department: "" });
  };

  const saveEmployee = async (id: number) => {
    setErrorMessage(null);
    setSuccessMessage(null);

    const trimmedName = editEmployee.name.trim();
    const trimmedEmail = editEmployee.email.trim();
    const trimmedDept = editEmployee.department.trim();

    if (!trimmedName) {
      setErrorMessage("Employee name cannot be blank.");
      return;
    }

    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
    if (!trimmedEmail || !emailRegex.test(trimmedEmail)) {
      setErrorMessage("Please enter a valid email address.");
      return;
    }

    if (!trimmedDept) {
      setErrorMessage("Department cannot be blank.");
      return;
    }

    try {
      setIsSaving(true);
      await API.put(`/employees/${id}`, {
        name: trimmedName,
        email: trimmedEmail,
        department: trimmedDept,
      });

      setSuccessMessage("Employee updated successfully.");
      setEditId(null);
      await loadEmployees(page);
    } catch (err: unknown) {
      let msg = "Failed to update employee.";
      if (axios.isAxiosError<ApiErrorData>(err)) {
        msg = err.response?.data?.message || err.response?.data?.error || msg;
      }
      setErrorMessage(msg);
    } finally {
      setIsSaving(false);
    }
  };

  // Wired to confirmDelete (DELETE /employees/{id})
  const confirmDelete = async () => {
    if (!deleteTarget) return;

    try {
      setIsDeleting(true);
      setErrorMessage(null);
      await API.delete(`/employees/${deleteTarget.id}`);

      setSuccessMessage(`Employee "${deleteTarget.name}" deleted successfully.`);
      setDeleteTarget(null);

      // If last item on current page (and not first page), go back one page
      if (employees.length === 1 && page > 0) {
        setPage((prev) => prev - 1);
      } else {
        await loadEmployees(page);
      }
    } catch (err: unknown) {
      let msg = "Failed to delete employee.";
      if (axios.isAxiosError<ApiErrorData>(err)) {
        msg = err.response?.data?.message || err.response?.data?.error || msg;
      }
      setErrorMessage(msg);
    } finally {
      setIsDeleting(false);
    }
  };

  // Filter employees based on search term (name, email, or department)
  const filteredEmployees = employees.filter((emp) => {
    const q = searchTerm.toLowerCase().trim();
    if (!q) return true;
    return (
      (emp.name && emp.name.toLowerCase().includes(q)) ||
      (emp.email && emp.email.toLowerCase().includes(q)) ||
      (emp.department && emp.department.toLowerCase().includes(q))
    );
  });

  return (
    <div className="card">
      <div className="dashboardHeader">
        <div>
          <h2>Employees</h2>
          <p className="cardSubtitle">Manage organization members and staff</p>
        </div>

        <input
          type="text"
          className="searchBar"
          placeholder="Search by name, email, department..."
          value={searchTerm}
          onChange={(e) => setSearchTerm(e.target.value)}
        />
      </div>

      {successMessage && (
        <div className="alert alertSuccess">{successMessage}</div>
      )}

      {errorMessage && <div className="alert alertError">{errorMessage}</div>}

      <div className="tableResponsive">
        <table className="employeeTable">
          <thead>
            <tr>
              <th>Name</th>
              <th>Email</th>
              <th>Department</th>
              {isAdmin && <th>Action</th>}
            </tr>
          </thead>

          <tbody>
            {loading && employees.length === 0 ? (
              <tr>
                <td colSpan={isAdmin ? 4 : 3} style={{ textAlign: "center", padding: "30px" }}>
                  Loading employees...
                </td>
              </tr>
            ) : filteredEmployees.length === 0 ? (
              <tr>
                <td colSpan={isAdmin ? 4 : 3} style={{ textAlign: "center", padding: "30px", opacity: 0.7 }}>
                  {searchTerm ? "No employees match your search." : "No employees found."}
                </td>
              </tr>
            ) : (
              filteredEmployees.map((emp) => (
                <tr key={emp.id}>
                  <td>
                    {editId === emp.id ? (
                      <input
                        className="inputField"
                        style={{ marginBottom: 0 }}
                        value={editEmployee.name}
                        onChange={(e) =>
                          setEditEmployee({ ...editEmployee, name: e.target.value })
                        }
                      />
                    ) : (
                      emp.name
                    )}
                  </td>

                  <td>
                    {editId === emp.id ? (
                      <input
                        type="email"
                        className="inputField"
                        style={{ marginBottom: 0 }}
                        value={editEmployee.email}
                        onChange={(e) =>
                          setEditEmployee({ ...editEmployee, email: e.target.value })
                        }
                      />
                    ) : (
                      emp.email
                    )}
                  </td>

                  <td>
                    {editId === emp.id ? (
                      <input
                        className="inputField"
                        style={{ marginBottom: 0 }}
                        value={editEmployee.department}
                        onChange={(e) =>
                          setEditEmployee({ ...editEmployee, department: e.target.value })
                        }
                      />
                    ) : (
                      emp.department
                    )}
                  </td>

                  {/* EDIT AND DELETE BUTTONS - ONLY VISIBLE TO ADMIN */}
                  {isAdmin && (
                    <td>
                      <div className="actionCell">
                        {editId === emp.id ? (
                          <>
                            <button
                              className="primaryButton"
                              style={{ padding: "6px 14px", fontSize: "13px" }}
                              onClick={() => saveEmployee(emp.id)}
                              disabled={isSaving}
                            >
                              {isSaving ? "Saving..." : "Save"}
                            </button>
                            <button
                              className="secondaryButton"
                              style={{ padding: "6px 12px", fontSize: "13px" }}
                              onClick={cancelEdit}
                              disabled={isSaving}
                            >
                              Cancel
                            </button>
                          </>
                        ) : (
                          <>
                            <button
                              className="primaryButton"
                              style={{ padding: "6px 14px", fontSize: "13px" }}
                              onClick={() => startEdit(emp)}
                            >
                              Edit
                            </button>
                            <button
                              className="deleteButton"
                              onClick={() => setDeleteTarget(emp)}
                            >
                              Delete
                            </button>
                          </>
                        )}
                      </div>
                    </td>
                  )}
                </tr>
              ))
            )}
          </tbody>
        </table>
      </div>

      {/* PAGINATION CONTROLS */}
      <div className="paginationControls">
        <button
          className="pageButton"
          onClick={() => setPage((prev) => Math.max(0, prev - 1))}
          disabled={page <= 0 || loading}
        >
          Previous
        </button>

        <span className="pageInfo">
          Page {totalPages > 0 ? page + 1 : 0} of {totalPages || 1} ({totalElements} total)
        </span>

        <button
          className="pageButton"
          onClick={() => setPage((prev) => Math.min(totalPages - 1, prev + 1))}
          disabled={page >= totalPages - 1 || loading || totalPages === 0}
        >
          Next
        </button>
      </div>

      {/* CONFIRM DELETE MODAL */}
      {deleteTarget && (
        <div className="modalOverlay" onClick={() => !isDeleting && setDeleteTarget(null)}>
          <div className="modalContent" onClick={(e) => e.stopPropagation()}>
            <h3 className="modalTitle">Confirm Deletion</h3>

              <button
                className="secondaryButton"

                Cancel

              >
                {isDeleting ? "Deleting..." : "Delete Employee"}
              </button>
            </div>
          </div>
        </div>
      )}
    </div>
  );
}

export default Employees;

import { useState, useEffect, type FormEvent } from "react";
import { useNavigate, Link, useLocation } from "react-router-dom";
import axios from "axios";
import API from "../api/api";
import { isAuthenticated } from "../utils/auth";
import "../styles/login.css";

interface ApiErrorData {
  message?: string;
  error?: string;
  errors?: Record<string, string>;
}

function Login() {
  const navigate = useNavigate();
  const location = useLocation();

  const [username, setUsername] = useState("");
  const [password, setPassword] = useState("");
  const [loading, setLoading] = useState(false);
  const [errorMessage, setErrorMessage] = useState<string | null>(null);
  const [successMessage, setSuccessMessage] = useState<string | null>(
    location.state?.message || null
  );

  useEffect(() => {
    if (isAuthenticated()) {
      navigate("/dashboard/employees", { replace: true });
    }
  }, [navigate]);

  const handleLogin = async (e: FormEvent) => {
    e.preventDefault();
    setErrorMessage(null);
    setSuccessMessage(null);

    if (!username.trim() || !password.trim()) {
      setErrorMessage("Please enter both username and password.");
      return;
    }

    try {
      setLoading(true);
      const res = await API.post("/auth/login", {
        username: username.trim(),
        password: password.trim(),
      });

      // Safe token extraction
      const token = res.data?.token ?? res.data;
      if (!token || typeof token !== "string") {
        throw new Error("Invalid token received from server.");
      }

      localStorage.setItem("token", token);
      navigate("/dashboard/employees");
    } catch (err: unknown) {
      let serverMessage = "Invalid username or password.";
      if (axios.isAxiosError<ApiErrorData>(err)) {
        serverMessage =
          err.response?.data?.message ||
          err.response?.data?.error ||
          err.message ||
          serverMessage;
      } else if (err instanceof Error) {
        serverMessage = err.message;
      }
      setErrorMessage(serverMessage);
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="loginContainer">
      <div className="card loginCard">
        <h1 className="loginTitle">Employee Portal</h1>
        <p className="loginSubtitle">Secure management system</p>

        {successMessage && (
          <div className="alert alertSuccess">{successMessage}</div>
        )}

        {errorMessage && <div className="alert alertError">{errorMessage}</div>}

        <form onSubmit={handleLogin}>
          <input
            className="inputField"
            placeholder="Username"
            value={username}
            onChange={(e) => setUsername(e.target.value)}
            disabled={loading}
            autoComplete="username"
            required
          />

          <input
            type="password"
            className="inputField"
            placeholder="Password"
            value={password}
            onChange={(e) => setPassword(e.target.value)}
            disabled={loading}
            autoComplete="current-password"
            required
          />

          <button
            type="submit"
            className="primaryButton"
            style={{ width: "100%" }}
            disabled={loading}
          >
            {loading ? "Signing In..." : "Sign In"}
          </button>
        </form>

          <span>Don't have an account? </span>
          <Link to="/signup" className="authLink">
            Sign Up
          </Link>
        </div>
      </div>
    </div>
  );
}

import { BrowserRouter, Routes, Route, Navigate } from "react-router-dom";
import Login from "./pages/Login";
import SignUp from "./pages/SignUp";
import Employees from "./pages/Employees";
import RegisterEmployee from "./pages/RegisterEmployee";
import Profile from "./pages/Profile";
import ManageUsers from "./pages/ManageUsers";
import DashboardLayout from "./layouts/DashboardLayout";
import ProtectedRoute from "./components/ProtectedRoute";
import AdminRoute from "./components/AdminRoute";

function App() {
  return (
    <BrowserRouter>
      <Routes>
        <Route path="/" element={<Login />} />
        <Route path="/signup" element={<SignUp />} />

        {/* Protected Dashboard Routes */}
        <Route
          path="/dashboard"
          element={
            <ProtectedRoute>
              <DashboardLayout />
            </ProtectedRoute>
          }
        >
          <Route index element={<Navigate to="/dashboard/employees" replace />} />
          <Route path="employees" element={<Employees />} />
          <Route path="profile" element={<Profile />} />

          {/* ADMIN-only Pages */}
          <Route element={<AdminRoute />}>
            <Route path="register" element={<RegisterEmployee />} />
            <Route path="users" element={<ManageUsers />} />
          </Route>
        </Route>

        {/* Catch-all route */}
        <Route path="*" element={<Navigate to="/" replace />} />
      </Routes>
    </BrowserRouter>
  );
}

export default App;
</file>
